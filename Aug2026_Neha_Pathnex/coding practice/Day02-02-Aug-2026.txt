# Ansible - Install & Enable Nginx
- name: Install and start Ngnix on Pathnex
  hosts: all
  become: yes


  tasks:
    - name: Install nginx
      yum:
	name: nginx
	state: present


    - name: Enable nginx
      service:
	name: nginx
	state: started
	enabled: yes

# Kubernetes - Deployment with 2 Replicas
 
apiVersion: app/v1
kind: Deployment
metadata:
   name: pathnex-deployment
spec:
  replicas: 2
  selector:
    matchLables:
	app: pathnex-app
  template:
    metadata:
	labels:
	   app: pathnex-app
    spec:
	containers:
	  -name: app
	   image: nginx
	   ports:
	     - containerPort: 80
